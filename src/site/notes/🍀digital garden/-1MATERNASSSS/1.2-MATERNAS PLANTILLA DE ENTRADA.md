`.trim();

/* ============================
PARSE
============================ */

function parseHeadersAndOptions(raw){

const lines=raw.split(/\n/);
const groups=[];
let current=null;

for(let line of lines){

line=line.trim();
if(!line)continue;

const m=line.match(/^header\s+(\d+):(.*)$/i);

if(m){

let title=m[2];
let slashes=0;

while(title.endsWith("/")){
slashes++;
title=title.slice(0,-1);
}

current={title:title.trim(),slashes,options:[]};
groups.push(current);

}else if(current){
current.options.push(line);
}

}

return groups;

}

/* ============================
UI
============================ */

function buildUI(){

const app=document.getElementById("app");
const groups=parseHeadersAndOptions(COMANDOS);

groups.forEach((g,i)=>{

const div=document.createElement("div");
div.className="grupo";
div.dataset.slashes=g.slashes;

const h=document.createElement("h3");
h.textContent=g.title;
div.appendChild(h);

g.options.forEach(opt=>{

const label=document.createElement("label");

const cb=document.createElement("input");
cb.type="checkbox";

label.appendChild(cb);
label.appendChild(document.createTextNode(opt));

div.appendChild(label);

});

app.appendChild(div);

});

}

/* ============================
NARRATIVA
============================ */

function updateNarrativa(){

let narrativa="";
const grupos=document.querySelectorAll(".grupo");

grupos.forEach(g=>{

const header=g.querySelector("h3").textContent;
const checked=g.querySelectorAll("input:checked");

if(checked.length){

narrativa+=header+" ";

checked.forEach(c=>{
narrativa+=c.parentElement.textContent+" ";
});

for(let i=0;i<g.dataset.slashes;i++){
narrativa+="\n\n";
}

}

});

const partes=narrativa.split("\n\n");

let html="";

partes.forEach(p=>{
html+="<p>"+p+"</p>";
});

document.getElementById("resultado").innerHTML=html;

}

buildUI();
