(function () {
  const SEP = "@@";
  const buttons = document.querySelectorAll(".ab11-buttons button");
  const panels = document.querySelectorAll(".ab11-panel");

  document.querySelectorAll(".ab11-panel pre code").forEach((code) => {
    const originalText = code.textContent.replace(/\r\n/g, "\n");
    code.dataset.copyText = originalText.replace(/^@@$/gm, "");

    const lines = originalText.split("\n");
    while (lines.length > 0 && lines[lines.length - 1] === "") lines.pop();

    const groups = [];
    let current = [];
    lines.forEach(line => {
      if (line.trim() === SEP) {
        if (current.length > 0) { groups.push({ text: current.join("\n"), sep: true }); current = []; }
      } else {
        current.push(line);
      }
    });
    if (current.length > 0) groups.push({ text: current.join("\n"), sep: false });

    code.innerHTML = "";
    groups.forEach((group) => {
      const span = document.createElement("span");
      span.className = "ab11-code-block";
      span.textContent = group.text;
      if (group.sep) span.style.marginBottom = "0.7em";
      code.appendChild(span);
    });
  });

  buttons.forEach((button) => {
    button.addEventListener("click", () => {
      const index = Number(button.dataset.tab);
      buttons.forEach((btn, i) => btn.classList.toggle("ab11-active", i === index));
      panels.forEach((panel, i) => panel.classList.toggle("ab11-active", i === index));
    });
  });

  document.querySelectorAll(".ab11-copy").forEach((button) => {
    button.addEventListener("click", () => {
      const code = button.parentElement.querySelector("code");
      navigator.clipboard.writeText(code.dataset.copyText).then(() => {
        button.textContent = "Copiado!";
        button.classList.add("copied");
        setTimeout(() => {
          button.textContent = "Copiar";
          button.classList.remove("copied");
        }, 1600);
      });
    });
  });
})();
</script>

<script>
(function () {
  const buttons = document.querySelectorAll(".ab22-buttons button");
  const panels = document.querySelectorAll(".ab22-panel");

  /*
    Procesa el contenido para:
    1. Respetar espacios y tabulaciones.
    2. Reducir la altura de líneas vacías.
    3. Mantener intacto el texto para copiar.
  */
  document.querySelectorAll(".ab22-panel pre code").forEach((code) => {
    let originalText = code.textContent.replace(/\r\n/g, "\n");

    /*
      Quitamos solo saltos de línea externos.
      No usamos trim(), porque trim() borra espacios y tabulaciones importantes.
    */
    originalText = originalText.replace(/^\n+|\n+$/g, "");

    code.dataset.copyText = originalText;

    const lines = originalText.split("\n");

    code.innerHTML = "";

    lines.forEach((line) => {
      const span = document.createElement("span");

      if (line === "") {
        span.className = "ab22-code-line ab22-empty-line";
        span.textContent = "";
      } else {
        span.className = "ab22-code-line";

        /*
          textContent + white-space: pre-wrap
          conserva espacios iniciales y tabulaciones.
        */
        span.textContent = line;
      }

      code.appendChild(span);
    });
  });

  buttons.forEach((button) => {
    button.addEventListener("click", () => {
      const index = Number(button.dataset.tab);

      buttons.forEach((btn, i) => {
        btn.classList.toggle("ab22-active", i === index);

(function () {
  const SEP = "@@";
  const buttons = document.querySelectorAll(".ab33-buttons button");
  const panels = document.querySelectorAll(".ab33-panel");

  document.querySelectorAll(".ab33-panel pre code").forEach((code) => {
    const originalText = code.textContent.replace(/\r\n/g, "\n").replace(/^\n+|\n+$/g, "");
    code.dataset.copyText = originalText.replace(/^@@$/gm, "");

    const lines = originalText.split("\n");
    while (lines.length > 0 && lines[lines.length - 1] === "") lines.pop();

    const groups = [];
    let current = [];
    lines.forEach(line => {
      if (line.trim() === SEP) {
        if (current.length > 0) { groups.push({ text: current.join("\n"), sep: true }); current = []; }
      } else {
        current.push(line);
      }
    });
    if (current.length > 0) groups.push({ text: current.join("\n"), sep: false });

    code.innerHTML = "";
    groups.forEach((group) => {
      const span = document.createElement("span");
      span.className = "ab33-code-block";
      span.textContent = group.text;
      if (group.sep) span.style.marginBottom = "0.7em";
